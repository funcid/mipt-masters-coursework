```sql
SET search_path TO stroy;

/**
Напишите запрос к заданию и укажите, было ли ежемесячное изменение 
полученной суммы по проектам больше чем на 150%.
Подсказка: с 2022-03-01 на 2022-04-01 изменение было 48,36%, 
с 2022-04-01 на 2022-05-01 изменение было 20,57%.
**/
WITH monthly_sum AS (
    SELECT
        DATE_TRUNC('month', fact_transaction_timestamp) AS month,
    SUM(amount) AS total_amount
FROM project_payment
WHERE fact_transaction_timestamp IS NOT NULL
GROUP BY 1
    ),
    monthly_change AS (
SELECT
    month,
    total_amount,
    LAG(total_amount) OVER (ORDER BY month) AS prev_amount,
    ROUND(
    (total_amount - LAG(total_amount) OVER (ORDER BY month))
    / LAG(total_amount) OVER (ORDER BY month) * 100,
    2
    ) AS percent_change
FROM monthly_sum
    )
SELECT *
FROM monthly_change;

/**
Напишите запрос, следуя шагам ниже, и укажите верный ответ.

Найдите скользящее среднее по стоимости проектов — текущее значение и две предыдущие записи. Найдите среднее значение полученных скользящих средних по всем проектам, округлённое до сотых (значение А).
Получите среднее значение стоимости всех проектов (значение Б).
Сравните значения А и Б. Какое среднее будет больше?
**/
WITH moving_avg AS (
    SELECT
        project_cost,
        AVG(project_cost) OVER (
            ORDER BY project_id
            ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
        ) AS rolling_avg
    FROM project
)
SELECT
    ROUND(AVG(rolling_avg), 2) AS value_A,
    ROUND(AVG(project_cost), 2) AS value_B,
    CASE
        WHEN ROUND(AVG(rolling_avg), 2) > ROUND(AVG(project_cost), 2) THEN 'A'
        WHEN ROUND(AVG(rolling_avg), 2) < ROUND(AVG(project_cost), 2) THEN 'B'
        ELSE 'Равны'
        END AS correct_answer
FROM moving_avg;

/**
Напишите запрос к заданию и выберите верный ответ.

Сделайте ранжирование проектов на каждый месяц по дате подписания. Чему будет равна сумма стоимости всех пятых проектов? Если в одно время подписано несколько проектов, то они все должны попасть в выборку.

Подсказка: всего 13 таких проектов.
**/
WITH ranked_projects AS (
    SELECT
        project_id,
        project_cost,
        DATE_TRUNC('month', sign_date) AS month,
    DENSE_RANK() OVER (
    PARTITION BY DATE_TRUNC('month', sign_date)
    ORDER BY sign_date
    ) AS rnk
FROM project
    )
SELECT
    COUNT(*) AS total_projects,
    SUM(project_cost) AS total_sum
FROM ranked_projects
WHERE rnk = 5;
```
