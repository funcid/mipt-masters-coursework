```sql
/**
Напишите запрос к заданию и выберите верный ответ: 
в каком структурном подразделении типа «Отдел», включая дочерние 
подразделения, работает больше всего сотрудников?
**/
WITH RECURSIVE dept_tree AS (
    -- Берём подразделения типа "Отдел"
    SELECT
        cs.unit_id,
        cs.unit_id AS root_unit_id
    FROM company_structure cs
             JOIN unit_type ut ON cs.unit_type = ut.unit_type_id
    WHERE ut.unit_type_name = 'Отдел'

    UNION ALL

    -- Добавляем дочерние подразделения
    SELECT
        cs.unit_id,
        dt.root_unit_id
    FROM company_structure cs
             JOIN dept_tree dt ON cs.parent_id = dt.unit_id
),
               dept_employee_count AS (
                   SELECT
                       dt.root_unit_id,
                       COUNT(DISTINCT ep.employee_id) AS emp_count
                   FROM dept_tree dt
                            JOIN position p ON p.unit_id = dt.unit_id
                            JOIN employee_position ep ON ep.position_id = p.position_id
                   GROUP BY dt.root_unit_id
               )
SELECT root_unit_id, emp_count
FROM dept_employee_count
ORDER BY emp_count DESC
    LIMIT 1;
```
