```sql
SET search_path TO stroy;

-- Из какой страны контрагент с идентификатором 100?
SELECT co.country_name
FROM customer cu
JOIN address a ON cu.address_id = a.address_id
JOIN city ci ON a.city_id = ci.city_id
JOIN country co ON ci.country_id = co.country_id
WHERE cu.customer_id = 100;

-- Как называется структурное подразделение, где числится руководитель, по проекту которого был совершён платёж с идентификатором 443?
SELECT cs.unit_name
FROM project_payment pp
JOIN project p ON pp.project_id = p.project_id
JOIN employee_position ep ON ep.employee_id = p.project_manager_id
JOIN position pos ON ep.position_id = pos.position_id
JOIN company_structure cs ON pos.unit_id = cs.unit_id
WHERE pp.project_payment_id = 443;

-- Как называется проект, которым управляет руководитель, проживающий в Туле?
SELECT p.project_name
FROM project p
JOIN employee e ON p.project_manager_id = e.employee_id
JOIN person per ON e.person_id = per.person_id
JOIN address a ON per.address_id = a.address_id
JOIN city c ON a.city_id = c.city_id
WHERE c.city_name = 'Тула';

-- Сколько сотрудников было нанято в те же дни, когда планировались платежи по проектам?
SELECT COUNT(*) AS employees_hired_on_payment_dates
FROM employee e
WHERE e.hire_date IN (
    SELECT DISTINCT pp.plan_payment_date
    FROM project_payment pp
);

-- На каком проекте не работает (не как руководитель проектов) сотрудник, у которого person_id = 35?
SELECT p.project_name
FROM project p
JOIN employee e ON e.employee_id = ANY(p.employees_id)
WHERE e.person_id = 35
  AND p.project_manager_id <> e.employee_id;

-- Какое количество сотрудников имеет самую большую з/п?
SELECT COUNT(*) AS num_employees
FROM employee_position ep
WHERE ep.salary = (
    SELECT MAX(salary)
    FROM employee_position
);

/*
  Какое количество фактических платежей было принято в октябре 2022 года по 
  проектам, где работали сотрудники (не руководители проектов) с 
  фамилией, начинающейся на К?
*/
SELECT COUNT(*) AS payments_count
FROM project_payment pp
WHERE pp.project_id IN (
    SELECT p.project_id
    FROM project p
    JOIN employee e ON e.employee_id = ANY(p.employees_id)
    JOIN person per ON e.person_id = per.person_id
    WHERE per.last_name LIKE 'К%'
      AND p.project_manager_id <> e.employee_id
)
AND pp.fact_transaction_timestamp >= '2022-10-01'
AND pp.fact_transaction_timestamp < '2022-11-01';

-- Какое количество сотрудников родилось в 1984 году и проживает в Одинцово?
SELECT COUNT(*) AS employees_count
FROM employee e
JOIN person p ON e.person_id = p.person_id
JOIN address a ON p.address_id = a.address_id
JOIN city c ON a.city_id = c.city_id
WHERE EXTRACT(YEAR FROM p.birthdate) = 1984
  AND c.city_name = 'Одинцово';

-- В каком структурном подразделении (значение идентификатора) работает наибольшее количество сотрудников?
SELECT pos.unit_id, COUNT(DISTINCT ep.employee_id) AS num_employees
FROM employee_position ep
JOIN position pos ON ep.position_id = pos.position_id
GROUP BY pos.unit_id
ORDER BY num_employees DESC
LIMIT 1;

-- Чему равно среднее значение дельты зарплатной вилки?
SELECT AVG(max_salary - min_salary) AS avg_delta
FROM grade_salary;
```
